use @std.time.{ms}
use @std.ui.components.{button, column, text}
use @std.ui.native.sources.{later}
use @std.ui.terminal.{mount}

state said: string = "waiting"

fn view() -> column {
    column {
        text "said ${said}"
        button "Ask" {
            on press {
                let _ = later(ms(150), () -> {
                    said = "answered"
                    true
                })
            }
        }
    }
}

mount(view)
println("said=${said}")
