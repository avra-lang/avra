use @std.ui.components.{button, column, field}
use @std.ui.terminal.{mount}

state name: string = ""
state hits: int = 0

fn view() -> column {
    column {
        field "Name" {
            value: name
            on input { typed -> name = typed }
        }
        button "Count" { on press { hits = hits + 1 } }
    }
}

mount(view)
println("name=[${name}] hits=${hits}")
