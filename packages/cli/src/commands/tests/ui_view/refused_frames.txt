use @std.meta.{derive}
use @std.ui.frame.derive.{Frame}

@derive(Frame)
enum Fine {
    @code(1) Bare
    @code(255) Last(n: int)
}

@derive(Frame)
enum Missing {
    @code(1) First
    Second
}

@derive(Frame)
enum Wide {
    @code(256) First
}

@derive(Frame)
enum Zero {
    @code(0) First
}

@derive(Frame)
enum Twice {
    @code(1) First
    @code(1) Second
}

@derive(Frame)
type Mapped = { m: Map<string, int> }

@derive(Frame)
enum Told {
    @code(1) First(@told n: int)
}

println("${Fine.frame_rows().length} ${Missing.frame_rows().length} ${Mapped.frame_rows().length}")
