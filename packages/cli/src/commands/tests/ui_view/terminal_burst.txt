use @std.time.{ms}
use @std.ui.components.{button, column, text}
use @std.ui.app.{later}
use @std.ui.terminal.{mount}

state bumped: int = 0

fn bump() -> bool {
    bumped = bumped + 1
    true
}

fn asks() {
    let _ = later(ms(100), bump)
    let _ = later(ms(100), bump)
    let _ = later(ms(100), bump)
}

fn view() -> column {
    column {
        text "bumped ${bumped}"
        button "Ask" { on press { asks() } }
    }
}

mount(view)
println("bumped=${bumped}")
