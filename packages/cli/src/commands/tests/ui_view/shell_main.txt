use @std.ui.components.{column}
use @std.ui.headless.{headless}
use parts.{shell}

fn page() -> column {
    column {
        shell "a"
        shell "b"
    }
}

let screen = headless(page)
let _ = screen.press("b+")
let _ = screen.press("b+")
println([l.trim() for l in screen.text().split("\n") if l.trim().starts_with("text")].join(" | "))
