use @std.meta.{derive}
use @std.ui.tree.prim.{prim}
use @std.ui.tree.view.{View}
use @std.ui.style.box.{Box}

@derive(View)
@prim(.Slider)
export component knob(@attr(.Label) label: string) {
    ..Box
    @attr(.Amount) amount: int = 0
    @host(web: .Change) on change(amount: int)
    @host(web: .Click, terminal: .Enter) on press
    @host(web: .Input) on input(typed: string)
    @host(web: .Submit) on submit
}
