use @std.meta.{derive}
use @std.ui.realize.prim.{prim}
use @std.ui.realize.view.{View}
use @std.ui.style.box.{Box}

@derive(View)
@prim(.Slider)
export component knob(@attr(.Label) label: string) {
    ..Box
    @attr(.Amount) amount: int = 0
    on change(amount: int)
    on press
    on input(typed: string)
    on submit
}
