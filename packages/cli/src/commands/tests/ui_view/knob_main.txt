use parts.{knob}
use @std.ui.realize.tests.support.{drawn_text}
use @std.ui.tree.view.{tree_of}

state turned: int = 0

fn dial() -> knob {
    knob "Volume" {
        amount: turned
        on change { amount -> turned = amount }
        on press { turned = 0 }
        on input { typed -> turned = typed.length }
        on submit { turned = 100 }
    }
}

println(drawn_text(tree_of(dial())))
