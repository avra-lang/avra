use @std.meta.{derive}
use @std.ui.realize.prim.{prim}
use @std.ui.realize.view.{View}
use @std.ui.style.box.{Box}
use @std.ui.style.tokens.{Tone}

@derive(View)
@prim(.Card)
export component gauge(@attr(.Title) title: string) {
    ..Box
    value: string = ""
    @attr(.Lable) other: string = ""
    @attr(.Level) words: string = ""
    @attr other_bare: string = ""
    @children kids: List<int> = []
    @children maybe: List<dyn View>? = null
    @style(colour) tone: string = ""
    @style(color) shade: string = ""
    @style plain: string = ""
    @attr(.Alt) @children both: string = ""
    @attr(.Label) on press
    on hover
    on input(amount: float)
}

@derive(View)
export component loose(@attr(.Title) title: string) {
    on press
}

@derive(View)
@prim(.Nope)
export component odd(@attr(.Title) title: string) {
    ..Box
}

@derive(View)
@prim(.Text)
export component echoing(@attr(.Text) content: string) {
    ..Box
    @attr(.Label) label: string = ""
    @attr(.Title) title: string = ""
    @attr(.Title) again: string = ""
    @style(color) tone: Tone = .Danger
    @style(color) other: Tone = .Muted
}
