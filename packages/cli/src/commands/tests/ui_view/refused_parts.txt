use @std.meta.{derive}
use @std.ui.tree.prim.{prim}
use @std.ui.tree.view.{View}
use @std.ui.components.{text}
use @std.ui.style.box.{Box}
use @std.ui.style.tokens.{Tone}

@derive(View)
@prim(.Card)
export component gauge(@attr(.Title) title: string) {
    ..Box
    value: string = ""
    state held: int = 0
    @attr(.Lable) other: string = ""
    @attr(.Level) words: string = ""
    @attr other_bare: string = ""
    @children kids: List<int> = []
    @children maybe: List<dyn View>? = null
    @style(colour) tone: string = ""
    @style(color) shade: string = ""
    @style plain: string = ""
    @attr(.Alt) @children both: string = ""
    @attr(.Label) on press
    on hover
    on input(amount: float)
}

@derive(View)
export component loose(title: string) {
    on press
}

@derive(View)
export component made(@attr(.Title) title: string) {
    ..Box
    text "made"
}

export component boxy(title: string) {
    ..Box
    text self.title
}

@derive(View)
@prim(.Text)
export component wordy(@attr(.Text) content: string) {
    ..Box
    text "wordy"
}

@derive(View)
@prim(.Text)
export component boxless(@attr(.Text) content: string) {}

@derive(View)
@prim(.Nope)
export component odd(@attr(.Title) title: string) {
    ..Box
}

@derive(View)
@prim(.Text)
export component echoing(@attr(.Text) content: string) {
    ..Box
    @attr(.Label) label: string = ""
    @attr(.Title) title: string = ""
    @attr(.Title) again: string = ""
    @style(color) tone: Tone = .Danger
    @style(color) other: Tone = .Muted
}
