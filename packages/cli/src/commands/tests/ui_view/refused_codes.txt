use @std.meta.{derive}
use @std.ui.realize.codes.{Codes}

@derive(Codes)
enum Fine {
    @code(0) Nothing
    @code(255) Last
}

@derive(Codes)
enum Missing {
    @code(0) First
    Second
}

@derive(Codes)
enum Worded {
    @code(abc) First
}

@derive(Codes)
enum Fraction {
    @code(1.5) First
}

@derive(Codes)
enum Wide {
    @code(300) First
}

@derive(Codes)
enum Negative {
    @code(-2) First
}

@derive(Codes)
enum Twice {
    @code(1) First
    @code(1) Second
}

@derive(Codes)
enum Carrying {
    @code(1) First(n: int)
}

println("${Fine.Last.code()} ${Fine.coded_as(0) != null} ${Fine.coded_as(7) == null}")
println("${Wide.First.code()} ${Missing.coded().length} ${Twice.coded_as(1) == null}")
