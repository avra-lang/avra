use @std.ui.tree.codes.{Codes}

@Codes
enum Fine {
    @code(0) Nothing
    @code(255) Last
}

@Codes
enum Missing {
    @code(0) First
    Second
}

@Codes
enum Worded {
    @code(abc) First
}

@Codes
enum Fraction {
    @code(1.5) First
}

@Codes
enum Wide {
    @code(300) First
}

@Codes
enum Negative {
    @code(-2) First
}

@Codes
enum Twice {
    @code(1) First
    @code(1) Second
}

@Codes
enum Carrying {
    @code(1) First(n: int)
}

println("${Fine.Last.code()} ${Fine.coded_as(0) != null} ${Fine.coded_as(7) == null}")
println("${Wide.First.code()} ${Missing.coded().length} ${Twice.coded_as(1) == null}")
