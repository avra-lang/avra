use @std.meta.{derive}
use @std.ui.tree.prim.{prim}
use @std.ui.tree.view.{View}
use @std.ui.components.{text}
use @std.ui.style.box.{Box}

@derive(View)
@prim(.Field)
export component good(@attr(.Label) label: string) {
    ..Box
    initial: string = ""
    @attr(.Value) state value: string = self.initial
    @over(value) shown: string? = null
    @sets(value) on input(typed: string)
    @sets(value) on paste(typed: string)
}

@derive(View)
@prim(.Field)
export component unseeded(@attr(.Label) label: string) {
    ..Box
    spare: string = ""
    @attr(.Value) state value: string = ""
    @sets(value) on input(typed: string)
}

@derive(View)
@prim(.Field)
export component overwide(@attr(.Label) label: string) {
    ..Box
    @attr(.Value) state value: string = ""
    @over(value) shown: int? = null
    @over(nothing) also: string? = null
    @sets(value) on input(typed: string)
}

@derive(View)
@prim(.Field)
export component overtwice(@attr(.Label) label: string) {
    ..Box
    @attr(.Value) state value: string = ""
    @over(value) shown: string? = null
    @over(value) again: string? = null
    @sets(value) on input(typed: string)
}

@derive(View)
@prim(.Checkbox)
export component missets(@attr(.Label) label: string) {
    ..Box
    @attr(.Checked) state checked: bool = false
    @sets(checked) on change(on: int)
    @sets(checked, 7) on reset
    @sets(gone) on clear
}

@derive(View)
@prim(.Checkbox)
export component unset(@attr(.Label) label: string) {
    ..Box
    @attr(.Checked) state checked: bool = false
    on change(on: bool)
}

component composed {
    state value: string = ""
    @over(value) shown: string? = null
    text "x"
}
