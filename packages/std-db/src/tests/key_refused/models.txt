use @std.db.{Many, model}

@model
type Tag = {
    id: int = 0,
    word: string,
}

@model
type Misnamed = {
    id: int = 0,
    name: string,
    @unique(nmae) slug: string,
}

@model
type OverMany = {
    id: int = 0,
    tags: Many<Tag>,
    @unique(tags) slug: string,
}

@model
type Doubled = {
    id: int = 0,
    name: string,
    @unique(name, name) slug: string,
}

@model
type OverItself = {
    id: int = 0,
    name: string,
    @unique(name, slug) slug: string,
}

@model
type Twice = {
    id: int = 0,
    name: string,
    @unique(name) @unique(name) slug: string,
}
