# UI tour — Boards, one program, every surface (v2)

> **Which model is built (2026-10-05):** this tour is the DESIGN's
> illustration. The library as built is `2026_10_05_UI_ARCHITECTURE.md`:
> components are imported from `@std.ui.components`, a handler is
> `on press { … }`, and an app is `mount(view)`. The running board is
> `tools/ui-board`.

> 2026-09-29. Illustrative. Design: `2026_09_29_UI.md` (§15 syntax, §16
> data). v2: one head/body rule, model-derived doors, data homes.

```avra
//! Boards — a team task board.
//! Web · iPhone · iPad · Android · Mac · terminal · widgets · email · Siri · agents · CLI.

use @std.ui.{app, screen, view, glance, product, theme, tokens, title, icon}
use @std.ui.{text, heading, image, badge, progress, avatars}
use @std.ui.{button, link, toggle, field, picker, form, section}
use @std.ui.{list, grid, table, row, column, scroll, spacer, split, detail, toolbar, menu, canvas, map, others}
use @std.camera.{camera}
use @std.notifications.{remind}
use @std.purchases.{buy}
use @std.mail.{mail}
use @std.time.{days, mins, Time, tomorrow}
use @std.ai.{agent, answer}
use @std.deploy.{deploy, secret}
use @std.testing.{render, fixture}
use @weather.{forecast}                 // a third-party API, typed from its OpenAPI at build (avra.toml)


// ══ 1. WORDS ═════════════════════════════════════════════════════════

enum Show { All Open Done }
enum Appearance { System Light Dark }

impl Show {
    fn admits(c: Card) -> bool {
        match self {
            .All -> true
            .Open -> !c.done
            .Done -> c.done
        }
    }
}


// ══ 2. DATA — each model has a home; its doors are derived from its policy ══

@model(access: .members, delete: .owner, trash: days(30), presence)
type Board = {
    name: string where it.length in 1..80,
    owner: User = me(),
    members: List<User> = [me()],
    site: Place?,
}

@model(access: .board, trash: days(7))                 // a card is its board's: same readers, same writers
type Card = {
    title: string where it.length in 1..200,
    done: bool = false,
    due: Time?,
    photo: Photo?,
    order: Rank,                                        // drag to reorder, derived
    board: Board,
}

@model(access: .board)
type Shape = { board: Board, kind: ShapeKind, at: Point, size: Size }

@model(at: .account)                                    // mine, on every device I sign into; never the board's
type Note = { card: Card @unique, text: string = "" }

@model(access: .board.owner)
type Invite = { board: Board, email: Email, accepted: bool = false }

impl Board {
    fn done_ratio() -> float { self.cards.ratio { it.done } }       // placed where the rows are
    fn home() -> Screen { board(self) }                             // where a board opens, everywhere
}

impl Card {
    fn home() -> Screen { board(self.board, open: self) }
}

fn due_soon() -> List<Card> { Card.all().where(done: false, due: .within(days(7))).sort(.due) }


// ══ 3. THE ONE HAND-WRITTEN DOOR — everything else is a model write ══

@action(auth: owner_of(b), impact: .external)
fn invite(b: Board, email: Email) -> Invite {
    let i = Invite.create(board: b, email: email)
    mail(email, invite_mail(i))                         // after commit · retried · receipted
    i
}


// ══ 4. STATE — a one-row model ═══════════════════════════════════════

state appearance: Appearance = .system at .account
state dense = false at .device


// ══ 5. LOOK AND MONEY ════════════════════════════════════════════════

theme brand {
    tokens: tokens(embed("design/tokens.json"))
    dark: tokens(embed("design/tokens.dark.json"))
}

product pro {                                           // → App Store, Play and Stripe products, at deploy
    monthly: usd(4)
    yearly: usd(40)
    paywall: upgrade()
}


// ══ 6. THE APP ═══════════════════════════════════════════════════════

app boards {
    name: "Boards"
    icon: embed("design/icon.svg")
    targets: [.web, .ios, .android, .macos, .terminal]
    appearance: appearance
    home()
    inbox()
    settings()
}


// ══ 7. SCREENS — head: the route · body: the page ════════════════════

screen home() {
    title "Boards"
    icon .grid
    toolbar {
        link("New board", to: form(Board), key: .primary + "n")
    }
    grid(Board.all())                                   // each by `view for Board`; a tap opens `home()`
    empty {
        link("New board", to: form(Board))
    }
}

screen board(b: Board, state show: Show = .all, state open: Card? = null) {
    title b.name
    toolbar {
        avatars(b.viewers)                              // presence, from `@model(presence)`
        picker(show)
        menu {
            link("Whiteboard", to: whiteboard(b))
            link("Invite", to: form(invite, b: b))
            link("Ask", to: ask(b))
            button("Delete board") { b.delete() }       // trash 30 d → undo, derived
        }
    }
    split {
        list(b.cards.filter { show.admits(it) }, selection: open)
        detail(open) { c -> card_detail(c) }
    }
    form(Card, board: b) {
        fields: [.title]
    }
}

screen inbox() {
    title "Due soon"
    icon .tray
    table(due_soon(), search: [.title, .board.name]) {  // search field · `?q=` · debounce — derived
        columns: [.title, .board, .due]
    }
}

screen whiteboard(b: Board) {
    title "${b.name} — whiteboard"
    requires: pro                                       // no entitlement → its paywall
    state cursor: Point? = null at .room(b)
    canvas(b.shapes) {                                  // the shapes are its semantic peer
        on frame(g) {
            for s in b.shapes { g.draw(s) }
            for p in others(cursor) { g.cursor(p) }
        }
        on hover(p) { cursor = p }
        on drag(s: Shape, p: Point) { s.at = p }        // a draft while dragging, one write on release
    }
}

screen settings() {
    title "Settings"
    icon .gear
    section("Look") {
        picker(appearance)                              // label from the name: "Appearance"
        toggle(dense)                                   // "Dense" — this device only
    }
    link("Boards Pro", to: upgrade())
}

screen upgrade() {
    title "Boards Pro"
    theme: brand
    column {
        gap: .l
        heading "Unlimited boards, whiteboards, history"
        for p in pro.plans { button(p.label) { buy(p)? } }   // "Purchasing…" until the store's receipt
        if compact { pad: .m }
    }
}

screen ask(b: Board, state prompt: string = "") {
    title "Ask"
    field(prompt, "Ask about ${b.name}")
    answer(helper, prompt, about: b)                    // a dyn View from the app's own views, checked whole
}

screen shared(b: Board, token: ShareToken) {
    targets: [.web]                                     // public · server HTML · 0 JS
    auth: shared_with(b, token)
    title b.name
    list(b.cards.filter { !it.done })                   // no write access → rows render read-only
}


// ══ 8. VIEWS — head: props · body: content, look, behaviour ══════════

view for Board {                                        // how a Board shows, wherever it shows
    column {
        pad: .m
        radius: .m
        fill: .surface_raised
        heading name
        progress done_ratio()
        if let s = site {
            map(s) { height: 120 }
            text forecast(s).summary                    // third-party, observed, cached
        }
    }
}

view for Card {
    row {
        gap: if dense { .xs } else { .s }
        toggle(done, title)                             // a model field is a place: the write is Card.update
        spacer
        badge(due?.relative()) {
            tone: if due?.past() ?? false { .danger } else { .muted }
        }
        if hovered { fill: .surface_hover }
    }
}

view card_detail(c: Card) {
    scroll {
        column {
            gap: .m
            heading c.title
            image(c.photo, alt: c.title)                // absent → nothing
            field(c.note.text, "Private note")           // my account's row, made on first write
            row {
                button("Photo") { c.photo = camera()? } // a pause · a prompt · an upload · a write
                button("Remind me") { remind(c.title, at: c.due ?? tomorrow(), open: c)? }
                link("Edit", to: form(c))               // a form over a row is a draft; conflicts merge per field
            }
            if platform is .ios {
                text "Shared with ${c.board.members.length}" { font: .footnote }
            }
        }
    }
}

view invite_mail(i: Invite) {                           // realized as email-safe HTML + plain text
    heading "Join ${i.board.name}"
    link("Open the board", to: i.board)                 // a typed deep link, in an email
}


// ══ 9. GLANCE — widget · complication · live activity ════════════════

glance today() {
    refresh: mins(15)
    heading "Due today"
    list(due_soon().take(3))                            // toggles write the device replica; the app syncs
    link("${due_soon().length} due", to: inbox())
}


// ══ 10. AGENTS ═══════════════════════════════════════════════════════

agent helper {
    model: "claude-sonnet-5"
    instructions: "Help the team keep boards tidy. Ask before deleting anything."
    tools: [Card.create, Card.update, Card.delete, invite]   // delete + invite: a signed mandate, never source
}


// ══ 11. PROOF ════════════════════════════════════════════════════════

spec "boards" {
    given "a board with three cards" {
        let b = fixture(Board, cards: 3)
        let first = b.cards[0]
        let ui = render(board(b))                       // the test realizer: headless
        then "a checked card is done before its receipt" {
            ui.toggle(first.title).press()
            ui.toggle(first.title).checked && ui.pending(first)
        }
        then "deleting the board offers undo" {
            ui.button("Delete board").press()
            ui.button("Undo").exists
        }
        then "a card opens where it lives" {
            render(first.home()).heading(first.title).exists
        }
    }
}


// ══ 12. MACHINES — the rest is derived ═══════════════════════════════

deploy production {
    region: "eu-west"
    domain: "boards.app"
    ios {
        bundle: "app.boards"
        team: secret("apple_team")
    }
    android { package: "app.boards" }
}
```

## Gaps the tour found

- `c.photo = camera()?` in a handler: which failures are announced, which are quiet (`Cancelled`) — a rule.
- `answer(...)` in render is an agent call — a pause and a cost; needs a cache/budget rule.
- `view for T` over a model: which fields a row may read without widening its replica (`c.board.members` in a row pulls members).
