use @std.meta.{Named, Type}

fn tag(_what: Named, _label: string) {}

type Entry = { kind: Type, name: string }

@tag("a")
type Widget = {
    label: string
    on: bool = false,
    size: int = 1,
}

@tag("b")
type Gadget = { label: string }

collect xs: List<Entry> = @tag in closure as Entry { kind: it, name: it.name } by it.name

fn fields_of(t: Type) -> int { t.fields.length }

const widths: List<int> = [fields_of(e.kind) for e in xs]

"${xs[0].name} ${widths[0]} ${xs[1].name} ${widths[1]}"
