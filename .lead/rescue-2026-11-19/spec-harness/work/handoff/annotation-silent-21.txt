--- main.av (as judged)
use @std.meta.{Trait, Kind, Diagnostic, refuse}
type Ctx = { n: int }
fn spell(k: Kind) -> string {
    match k {
        .Var(w) -> "Var ${w}",
        .Named(w, _) -> "Named ${w}",
        .Int -> "Int",
        rest -> "other",
    }
}
fn probe(tr: Trait) -> List<Diagnostic> {
    [refuse("kinds ${[spell(p.kind) for p in f.params].join(",")}") for f in tr.fns]
}
@probe
trait T {
    fn a(cx: Ctx, o: Self, n: int) -> Self
}
