### base_flow
@mutate
fn pick(x: int?) -> int {
    let v? = x else { return 0 }
    if v > 1 { v } else { 1 }
}

fn count(xs: List<int>) -> int {
    mut n = 0
    for x in xs {
        if x > 0 { n = n + 1 }
    }
    n
}

count([1, 2, 3]) + pick(null)
